-- ============================
-- CREAR BASE DE DATOS
-- ============================
CREATE DATABASE tp_poo;
USE tp_poo;

-- ============================
-- TABLA USUARIO
-- ============================
CREATE TABLE usuario (
    idUsuario INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    apellido VARCHAR(100) NOT NULL,
    email VARCHAR(150) NOT NULL UNIQUE,
    contrasena VARCHAR(255) NOT NULL,
    tipoUsuario ENUM('ADMIN', 'ALUMNO', 'DOCENTE') NOT NULL
);

-- ============================
-- TABLA ALUMNO
-- ============================
CREATE TABLE alumno (
    idUsuario INT PRIMARY KEY,
    legajo VARCHAR(50) NOT NULL UNIQUE,
    FOREIGN KEY (idUsuario) REFERENCES usuario(idUsuario)
        ON DELETE CASCADE
        ON UPDATE CASCADE
);

-- ============================
-- TABLA DOCENTE
-- ============================
CREATE TABLE docente (
    idUsuario INT PRIMARY KEY,
    matricula VARCHAR(50) NOT NULL UNIQUE,
    FOREIGN KEY (idUsuario) REFERENCES usuario(idUsuario)
        ON DELETE CASCADE
        ON UPDATE CASCADE
);

-- ============================
-- TABLA AREA
-- ============================
CREATE TABLE area (
    idArea INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL UNIQUE
);

-- ============================
-- TABLA CURSO
-- ============================
CREATE TABLE curso (
    idCurso INT AUTO_INCREMENT PRIMARY KEY,
    titulo VARCHAR(150) NOT NULL,
    cupoMax INT NOT NULL CHECK (cupoMax > 0),
    idDocente INT NOT NULL,
    idArea INT NOT NULL,
    contenido TEXT,
    activo BOOLEAN DEFAULT TRUE
    cantidadClases INT NOT NULL CHECK (cantidadClases > 0),

    FOREIGN KEY (idDocente) REFERENCES docente(idUsuario)
        ON DELETE RESTRICT
        ON UPDATE CASCADE,

    FOREIGN KEY (idArea) REFERENCES area(idArea)
        ON DELETE RESTRICT
        ON UPDATE CASCADE
);

-- ============================
-- TABLA INSCRIPCION
-- ============================
CREATE TABLE inscripcion (
    idInscripcion INT AUTO_INCREMENT PRIMARY KEY,
    fecha DATETIME NOT NULL,
    idUsuario INT NOT NULL,
    idCurso INT NOT NULL,
    idPago INT NULL,

    estadoPago ENUM('PENDIENTE_PAGO','PAGO') NOT NULL,
    estadoCurso ENUM('CURSANDO','DADO_DE_BAJA','APROBADO','DESAPROBADO') NOT NULL,

    FOREIGN KEY (idUsuario) REFERENCES alumno(idUsuario)
        ON DELETE CASCADE
        ON UPDATE CASCADE,

    FOREIGN KEY (idCurso) REFERENCES curso(idCurso)
        ON DELETE CASCADE
        ON UPDATE CASCADE
);

-- ============================
-- TABLA PAGO
-- ============================
CREATE TABLE pago (
    idPago INT AUTO_INCREMENT PRIMARY KEY,
    fecha DATETIME NOT NULL,
    monto DECIMAL(10,2) NOT NULL CHECK (monto > 0),
    idUsuario INT NOT NULL,

    FOREIGN KEY (idUsuario) REFERENCES alumno(idUsuario)
        ON DELETE CASCADE
        ON UPDATE CASCADE
);

-- ============================
-- TABLA CLASE
-- ============================
CREATE TABLE clase (
    idClase INT AUTO_INCREMENT PRIMARY KEY,
    idCurso INT NOT NULL,
    fecha DATETIME NOT NULL,
    titulo VARCHAR(150) NOT NULL,
    contenido TEXT NOT NULL,

    FOREIGN KEY (idCurso) REFERENCES curso(idCurso)
        ON DELETE CASCADE
        ON UPDATE CASCADE
);

-- ============================
-- TABLA ASISTENCIA
-- ============================
CREATE TABLE asistencia (
    idAsistencia INT AUTO_INCREMENT PRIMARY KEY,
    idInscripcion INT NOT NULL,
    idClase INT NOT NULL,
    presente BOOLEAN NOT NULL DEFAULT FALSE,

    FOREIGN KEY (idInscripcion) REFERENCES inscripcion(idInscripcion)
        ON DELETE CASCADE,

    FOREIGN KEY (idClase) REFERENCES clase(idClase)
        ON DELETE CASCADE,

    UNIQUE (idInscripcion, idClase)
);

-- ============================
-- TABLA CALIFICACION
-- ============================
CREATE TABLE calificacion (
    idCalificacion INT AUTO_INCREMENT PRIMARY KEY,
    idInscripcion INT NOT NULL,
    tipo ENUM('PARCIAL','FINAL','TRABAJO_PRACTICO','RECUPERATORIO','ACTIVIDAD_EN_CLASE') NOT NULL,
    nota DECIMAL(4,2) NOT NULL CHECK (nota >= 0 AND nota <= 10),
    fecha DATETIME NOT NULL,

    FOREIGN KEY (idInscripcion) REFERENCES inscripcion(idInscripcion)
        ON DELETE CASCADE
        ON UPDATE CASCADE
);
